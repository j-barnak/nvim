        mov      ebp, esp
        and      esp, 0FFFFFFF0h
        sub      esp, 10h
        fld      ds:dword_8048614 ; -4.936
        fstp     qword ptr [esp+8]
        mov      dword ptr [esp+4], offset asc_8048610 ; "%f\n"
        mov      dword ptr [esp], 1
        call     ___printf_chk
        xor      eax, eax
        leave
        retn
main    endp

The f() function is almost understandable. However, what is interesting is that
GCC was able to calculate the result of f(1.234) during compilation despite all this
hodge-podge with the structure fields and prepared this argument to printf() as
precalculated at compile time!

1.31      The classic struct bug
This is a classic struct bug.
A sample definition:
struct test
{
        int field1;
        int field2;
};

And then C files:
void setter(struct test *t, int a, int b)
{
        t->field1=a;
        t->field2=b;
};

#include <stdio.h>

void printer(struct test *t)
{
        printf ("%d\n", t->field1);
        printf ("%d\n", t->field2);
};
